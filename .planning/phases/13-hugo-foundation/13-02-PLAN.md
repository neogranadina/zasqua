From eleventy.config.js:37-64 (formatDateNarrative — D-15, D-16):
```js
function formatDateNarrative(dateStr) {
  if (!dateStr) return '';
  const SPANISH_MONTHS = ['enero','febrero','marzo','abril','mayo','junio',
                          'julio','agosto','septiembre','octubre','noviembre','diciembre'];
  if (dateStr.indexOf(' .. ') !== -1) {
    const [start, end] = dateStr.split(' .. ');
    return formatDateNarrative(start) + ' – ' + formatDateNarrative(end);
  }
  const ymd = dateStr.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (ymd) return `${parseInt(ymd[3],10)} de ${SPANISH_MONTHS[parseInt(ymd[2],10)-1]} de ${ymd[1]}`;
  const ym = dateStr.match(/^(\d{4})-(\d{2})$/);
  if (ym) return `${SPANISH_MONTHS[parseInt(ym[2],10)-1]} de ${ym[1]}`;
  return dateStr;
}
```
The range join character is EM-DASH-ish — specifically U+2013 EN DASH surrounded by ASCII spaces (" – "). Not hyphen.

From eleventy.config.js:68-71 (numberFormat — D-17):
```js
function numberFormat(num) {
  if (num === null || num === undefined) return '0';
  return num.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}
```
Spanish thousands separator is `.` (period), not `,` (comma). Test cases: 1000 → "1.000", 106529 → "106.529", 0 → "0", null → "0".

From src/_data/descriptions.js (ancestor walker — D-08):
```js
function buildAncestorChain(desc, byRefCode) {
  const chain = [];
  let current = desc;
  while (current && current.parent_reference_code) {
    const parent = byRefCode.get(current.parent_reference_code);
    if (!parent) break;
    chain.unshift({
      reference_code: parent.reference_code,
      title: parent.title,
      description_level: parent.description_level,
    });
    current = parent;
  }
  return chain;
}
```
Add a depth cap of 20 with an errorf-equivalent console.error + `process.exit(1)` — per Research §Code Examples note. Cycle in source data is a backend bug that must not be silenced.
