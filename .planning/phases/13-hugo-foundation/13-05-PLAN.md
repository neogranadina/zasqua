<!-- New build.sh skeleton — replaces the Eleventy build pipeline -->

build.sh STRUCTURE:
```bash
#!/usr/bin/env bash
#
# Zasqua frontend build orchestrator (Hugo edition)
#
# This script runs the full build pipeline for the static site. It does four
# things, in order:
#
#   1. (optional) Download the latest B2 exports into exports/
#   2. Run precompute-links.js to produce the entity/place link shards
#      that client-side JS will fetch at runtime
#   3. Run generate-content.js to enrich descriptions, entities, and places
#      and write flat JSON into assets/hugo-data/ for Hugo to consume
#   4. Run hugo --minify to produce the final public/ directory
#
# DEV_LIMIT environment variable: when set, generate-content.js truncates
# each dataset to that many records; Hugo then renders only that subset.
# Use DEV_LIMIT=100 for a fast smoke build (seconds, not minutes).
#
# SKIP_DOWNLOAD=1 skips the B2 download step — useful when iterating on
# enrichment or templates against already-downloaded data.
#
# Version: v1.0.0
#
set -euo pipefail

# --- Preconditions ---
hugo version | grep -q "+extended" || { echo "ERROR: Hugo Extended is required (run: brew install hugo)" >&2; exit 1; }

# --- B2 download (skippable) ---
if [ "${SKIP_DOWNLOAD:-0}" != "1" ]; then
  echo "[build] downloading B2 exports to exports/"
  # existing b2 sync invocation — PATH from previous build.sh with data/ → exports/
  # (preserve whatever auth + bucket logic was already here)
  pip install --quiet 'b2[full]'
  b2 sync --delete b2://zasqua-export exports/
fi

# --- Precompute link shards ---
echo "[build] running precompute-links"
node scripts/precompute-links.js

# --- Populate static/data/ with client-runtime shards ---
# Client JS fetches /data/children/*.json, /data/entity-links/*.json, etc.
# Hugo's static/ passthrough serves them. Copy (don't symlink — symlinks are
# fragile in Hugo passthrough).
echo "[build] copying runtime shards to static/data/"
rm -rf static/data
mkdir -p static/data
cp -r exports/children static/data/children
cp -r exports/entity-links static/data/entity-links
cp -r exports/place-links static/data/place-links
cp exports/entity-index.json static/data/entity-index.json
cp exports/place-index.json static/data/place-index.json
cp exports/curated-entity-graph.json static/data/curated-entity-graph.json 2>/dev/null || true

# --- Enrichment ---
echo "[build] running generate-content"
if [ -n "${DEV_LIMIT:-}" ]; then
  DEV_LIMIT="$DEV_LIMIT" node scripts/generate-content.js
else
  node scripts/generate-content.js
fi

# --- Hugo build ---
echo "[build] running hugo --minify"
hugo --minify

echo "[build] done — output in public/"
```
