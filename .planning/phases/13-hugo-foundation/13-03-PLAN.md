```toml
# file header comment per code-conventions.md: narrative header + Version footer
# (toml uses # comments)

baseURL = "https://zasqua.org/"
languageCode = "es-CO"
defaultContentLanguage = "es"
title = "Zasqua — Archivo Histórico"

[build]
  writeStats = true
  [build.buildStats]
    enable = true

# Module-mount pattern for hugo_stats.json — see RESEARCH.md Pitfall 2.
# Tailwind v4 silently skips .gitignored files; this mount makes hugo_stats.json
# visible under assets/ where Tailwind's @source directive can reach it.
[module]
  [[module.mounts]]
    source = "assets"
    target = "assets"
  [[module.mounts]]
    source = "hugo_stats.json"
    target = "assets/notwatching/hugo_stats.json"
    disableWatch = true

[[build.cachebusters]]
  source = 'assets/notwatching/hugo_stats\.json'
  target = 'css'

[[build.cachebusters]]
  source = '(postcss|tailwind)\.config\.js'
  target = 'css'

[markup]
  [markup.goldmark.renderer]
    unsafe = true

# Footer: Version: v1.0.0
```

DESCRIPTIONS ADAPTER (content/descripcion/_content.gotmpl):
```gotmpl
{{/*
  Zasqua descriptions content adapter

  This template is Hugo's way of turning a large JSON file into a page per
  record without ever writing stub markdown to disk. It reads the enriched
  descriptions JSON written by scripts/generate-content.js at build time
  and emits one Hugo page for each record, rooted at /{reference_code}/
  per D-18.

  The try/.Err/errorf triple is non-negotiable — bare resources.Get returns
  nil on missing files and would silently produce an empty build.

  Source: https://gohugo.io/content-management/content-adapters/

  Version: v1.0.0
*/}}
{{ $data := slice }}
{{ with try (resources.Get "hugo-data/descriptions.json") }}
  {{ with .Err }}
    {{ errorf "[descripcion adapter] Unable to read hugo-data/descriptions.json: %s" . }}
  {{ else }}
    {{ $data = .Value | transform.Unmarshal }}
  {{ end }}
{{ end }}

{{ range $data }}
  {{ $params := dict
      "reference_code" .reference_code
      "parent_reference_code" .parent_reference_code
      "repository_code" .repository_code
      "repository" .repository
      "ancestor_chain" .ancestor_chain
      "date_formatted" .date_formatted
      "entity_links" .entity_links
      "place_links" .place_links
      "description_level" .description_level
  }}
  {{ $page := dict
      "kind" "page"
      "path" .reference_code
      "title" .title
      "params" $params
      "content" (dict "mediaType" "text/html" "value" "")
  }}
  {{ $page = merge $page (dict "url" (printf "/%s/" .reference_code)) }}
  {{ $.AddPage $page }}
{{ end }}
```
