Year range implementation:
```gotmpl
{{ with .Params.date_earliest }}
  {{ $s := int . }}
  {{ $e := int (default $s $.Params.date_latest) }}
  {{ if gt $e (add $s 500) }}{{ $e = (add $s 500) }}{{ end }}
  {{ range $y := seq $s $e }}
    <span data-pagefind-filter="year">{{ $y }}</span>
  {{ end }}
{{ end }}
```

Century range:
```gotmpl
{{ $startCentury := add (math.Floor (math.Div (sub $s 1) 100) | int) 1 }}
{{ $endCentury := add (math.Floor (math.Div (sub $e 1) 100) | int) 1 }}
{{ range $c := seq $startCentury $endCentury }}
  <span data-pagefind-filter="century">{{ $c }}</span>
{{ end }}
```

Decade range:
```gotmpl
{{ $startDecade := mul (math.Floor (math.Div $s 10) | int) 10 }}
{{ $endDecade := mul (math.Floor (math.Div $e 10) | int) 10 }}
{{ range $d := seq $startDecade 10 $endDecade }}
  <span data-pagefind-filter="decade">{{ $d }}</span>
{{ end }}
```
