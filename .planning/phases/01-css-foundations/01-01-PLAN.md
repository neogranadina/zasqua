```bash
echo "=== Building CSS with Tailwind ==="
ARCH=$(uname -m)
if [ "$ARCH" = "arm64" ]; then
  TW_BINARY="tailwindcss-macos-arm64"
elif [ "$ARCH" = "x86_64" ] && [ "$(uname -s)" = "Darwin" ]; then
  TW_BINARY="tailwindcss-macos-x64"
else
  TW_BINARY="tailwindcss-linux-x64"
fi
if [ ! -f ./tailwindcss ]; then
  curl -sLO "https://github.com/tailwindlabs/tailwindcss/releases/latest/download/$TW_BINARY"
  chmod +x "$TW_BINARY"
  mv "$TW_BINARY" tailwindcss
fi
./tailwindcss -i src/css/input.css -o src/css/main.css --minify
```

**4. Create `scripts/check-css-tokens.sh`** — automated verification script for COL-01 through COL-04:

```bash
#!/bin/bash
# CSS token verification for Phase 1.
# Checks that the compiled main.css has correct tokens and no legacy colours.
set -e

CSS="src/css/main.css"
ERRORS=0

echo "=== CSS Token Verification ==="

# COL-01: Brand tokens present
for token in color-burgundy color-burgundy-deep color-burgundy-light color-burgundy-dark \
             color-pale-rose color-ochre color-sage color-periwinkle color-bg; do
  if grep -q "$token" "$CSS"; then
    echo "PASS: $token found"
  else
    echo "FAIL: $token NOT found"
    ERRORS=$((ERRORS + 1))
  fi
done

# COL-03: No blue accent colours
BLUE_COUNT=$(grep -c "41,98,255\|2c3e50\|003660" "$CSS" || true)
if [ "$BLUE_COUNT" -eq 0 ]; then
  echo "PASS: No blue accent colours"
else
  echo "FAIL: $BLUE_COUNT blue accent colour(s) remain"
  ERRORS=$((ERRORS + 1))
fi

# COL-04: No orange hover colours
ORANGE_COUNT=$(grep -ci "f18e00\|F2784B" "$CSS" || true)
if [ "$ORANGE_COUNT" -eq 0 ]; then
  echo "PASS: No orange hover colours"
else
  echo "FAIL: $ORANGE_COUNT orange hover colour(s) remain"
  ERRORS=$((ERRORS + 1))
fi

echo ""
if [ "$ERRORS" -eq 0 ]; then
  echo "All checks passed."
  exit 0
else
  echo "$ERRORS check(s) failed."
  exit 1
fi
```

Make it executable: `chmod +x scripts/check-css-tokens.sh`
