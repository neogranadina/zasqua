**`computeHopDistances()` method:**
BFS from `this.focalEntityCode` over `this.nodeNeighbours`:
```javascript
var hopMap = new Map();
hopMap.set(this.focalEntityCode, 0);
var queue = [this.focalEntityCode];
while (queue.length) {
  var curr = queue.shift();
  var currHop = hopMap.get(curr);
  var neighbours = this.nodeNeighbours.get(curr) || new Set();
  neighbours.forEach(function(nb) {
    if (!hopMap.has(nb)) {
      hopMap.set(nb, currHop + 1);
      queue.push(nb);
    }
  });
}
this.hopDistance = hopMap;
```
