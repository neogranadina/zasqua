**Structure:**
```javascript
document.addEventListener('DOMContentLoaded', async () => {
  const timelineEl = document.getElementById('entity-timeline');
  if (!timelineEl) return;

  const entityCode = timelineEl.dataset.entityCode;
  if (!entityCode) return;

  try {
    const res = await fetch(`/data/entity-links/${entityCode}.json`);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const links = await res.json();

    renderTimeline(timelineEl, links, true); // true = show role labels
  } catch (err) {
    console.error('[entity] Failed to load shard:', err);
    // Use the error message from ui.js — but since this is client-side JS
    // and ui.js is only available at build time, hardcode the Spanish string:
    timelineEl.innerHTML = '<p class="text-stone-500 text-sm">No se pudieron cargar las descripciones vinculadas. Intente recargar la página.</p>';
  }
});
```

**renderTimeline function:**
```javascript
function renderTimeline(container, links, showRoles) {
  if (!links || links.length === 0) {
    container.innerHTML = '<p class="text-stone-500 text-sm">No se encontraron descripciones vinculadas a este registro.</p>';
    return;
  }

  // Role labels map (matching ui.roles from ui.js)
  const roleLabels = {
    creator: 'Productor',
    contributor: 'Colaborador',
    publisher: 'Editor',
    subject: 'Materia',
    mentioned: 'Mencionado'
  };

  // Separate dated and undated entries
  const dated = [];
  const undated = [];
  for (const link of links) {
    if (link.date_expression) {
      dated.push(link);
    } else {
      undated.push(link);
    }
  }

  // Sort dated entries by date_expression (lexicographic — dates are normalised year strings)
  dated.sort((a, b) => a.date_expression.localeCompare(b.date_expression));

  // Build HTML
  let html = '';

  for (const link of dated) {
    html += renderTimelineEntry(link, showRoles, roleLabels);
  }

  if (undated.length > 0) {
    html += '<div class="timeline-no-date">Sin fecha</div>';
    for (const link of undated) {
      html += renderTimelineEntry(link, showRoles, roleLabels);
    }
  }

  container.innerHTML = html;

  // Update the linked descriptions count in the CTA link
  const ctaEl = document.getElementById('linked-desc-cta');
  if (ctaEl) {
    // The template renders the count from _linked_count at build time,
    // but the shard may have a different count. Update with shard length.
    const countText = ctaEl.textContent.replace(/\d+/, links.length);
    ctaEl.textContent = countText;
  }
}
```

**renderTimelineEntry helper:**
```javascript
function renderTimelineEntry(link, showRoles, roleLabels) {
  // Build the description page URL using safeSlug pattern (remove ? and #)
  const slug = link.reference_code.replace(/[?#]/g, '');
  let html = '<div class="timeline-entry">';

  if (link.date_expression) {
    html += `<div class="timeline-date">${escapeHtml(link.date_expression)}</div>`;
  }

  if (showRoles && link.role) {
    const roleLabel = roleLabels[link.role] || link.role;
    html += `<div class="timeline-role">${escapeHtml(roleLabel)}</div>`;
  }

  html += `<a href="/${slug}/" class="timeline-title">${escapeHtml(link.title)}</a>`;
  html += '</div>';
  return html;
}
```

**escapeHtml helper (XSS prevention):**
```javascript
function escapeHtml(str) {
  if (!str) return '';
  return str.replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;');
}
```

**Key details:**
- Description page links use the same `safeSlug` pattern as the Eleventy build: `/{reference_code}/` with `?` and `#` stripped
- Role labels are hardcoded in JS since `ui.js` is only available at build time, not client-side. Use the exact same values as `ui.roles` in ui.js: `{ creator: 'Productor', contributor: 'Colaborador', publisher: 'Editor', subject: 'Materia', mentioned: 'Mencionado' }`
- Timeline entries for entity pages always show role labels (D-16)
- "Sin fecha" group appears at the bottom (D-15)
- Container scrolls via CSS `.entity-timeline` max-height (D-14 — no JS-based truncation)
