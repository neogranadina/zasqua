**Structure:**
```javascript
document.addEventListener('DOMContentLoaded', async () => {
  // 1. Initialise map (if map element exists)
  initMap();

  // 2. Fetch and render timeline
  const timelineEl = document.getElementById('place-timeline');
  if (!timelineEl) return;

  const placeCode = timelineEl.dataset.placeCode;
  if (!placeCode) return;

  try {
    const res = await fetch(`/data/place-links/${placeCode}.json`);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const links = await res.json();

    renderTimeline(timelineEl, links, false); // false = no role labels for place pages
  } catch (err) {
    console.error('[place] Failed to load shard:', err);
    timelineEl.innerHTML = '<p class="text-stone-500 text-sm">No se pudieron cargar las descripciones vinculadas. Intente recargar la página.</p>';
  }
});
```

**initMap function (per D-09, D-10, D-11):**
```javascript
function initMap() {
  const mapEl = document.getElementById('place-map');
  if (!mapEl) return; // No coordinates — no map

  const lat = parseFloat(mapEl.dataset.lat);
  const lon = parseFloat(mapEl.dataset.lon);
  if (isNaN(lat) || isNaN(lon)) return;

  // pmtiles protocol already registered by inline module script in <head>
  // Do NOT call addProtocol here (Pitfall 3 from RESEARCH.md)

  const map = new maplibregl.Map({
    container: 'place-map',
    style: 'https://cdn.protomaps.com/basemaps/v4/en.json',
    center: [lon, lat],
    zoom: 7
  });

  // Burgundy pin marker (12px circle, #8B2942, 2px white border)
  const markerEl = document.createElement('div');
  markerEl.style.width = '12px';
  markerEl.style.height = '12px';
  markerEl.style.borderRadius = '50%';
  markerEl.style.backgroundColor = '#8B2942';
  markerEl.style.border = '2px solid #FFFFFF';

  new maplibregl.Marker({ element: markerEl })
    .setLngLat([lon, lat])
    .addTo(map);
}
```

**About the Protomaps basemap style:** The `https://cdn.protomaps.com/basemaps/v4/en.json` style is a complete style with its own tile sources (Protomaps raster tiles). The PMTiles protocol handler + `tiles.zasqua.org` is NOT used as the basemap — that's for the heatmap layer in Phase 7. For the detail page embed, the Protomaps default light basemap provides geographic context. The marker pin shows the place's location. This is correct per D-10 (country/region-level zoom for orientation).

**renderTimeline function (same as entity.js but without role labels):**
```javascript
function renderTimeline(container, links, showRoles) {
  if (!links || links.length === 0) {
    container.innerHTML = '<p class="text-stone-500 text-sm">No se encontraron descripciones vinculadas a este registro.</p>';
    return;
  }

  const dated = [];
  const undated = [];
  for (const link of links) {
    if (link.date_expression) {
      dated.push(link);
    } else {
      undated.push(link);
    }
  }

  dated.sort((a, b) => a.date_expression.localeCompare(b.date_expression));

  let html = '';
  for (const link of dated) {
    html += renderTimelineEntry(link, showRoles);
  }
  if (undated.length > 0) {
    html += '<div class="timeline-no-date">Sin fecha</div>';
    for (const link of undated) {
      html += renderTimelineEntry(link, showRoles);
    }
  }

  container.innerHTML = html;

  // Update CTA count from shard
  const ctaEl = document.getElementById('linked-desc-cta');
  if (ctaEl) {
    const countText = ctaEl.textContent.replace(/\d+/, links.length);
    ctaEl.textContent = countText;
  }
}

function renderTimelineEntry(link, showRoles) {
  const slug = link.reference_code.replace(/[?#]/g, '');
  let html = '<div class="timeline-entry">';
  if (link.date_expression) {
    html += `<div class="timeline-date">${escapeHtml(link.date_expression)}</div>`;
  }
  // Place timelines do NOT show role labels (per D-13 — role only for entity pages)
  html += `<a href="/${slug}/" class="timeline-title">${escapeHtml(link.title)}</a>`;
  html += '</div>';
  return html;
}

function escapeHtml(str) {
  if (!str) return '';
  return str.replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;');
}
```
